def hp(y,l):
 o=[]
 for t in range(1,len(y)+1):
  d=np.diff(np.eye(t),2,axis=0);o.append(y[t-1] if t<3 else np.linalg.solve(np.eye(t)+l*d.T@d,y[:t])[-1])
